def resolver_a_estrella(x_ini, y_ini):
    ruta = [(x_ini, y_ini)]
    visitados = {(x_ini, y_ini)}

    def backtrack(x, y, paso):
        if paso == 64:
            return True
        
        candidatos = []
        for nx, ny in obtener_movimientos(x, y):
            if (nx, ny) not in visitados:
                g = paso
                h = sum(1 for vx, vy in obtener_movimientos(nx, ny) if (vx, vy) not in visitados)
                f = g + h
                candidatos.append((f, h, nx, ny))
        
        candidatos.sort(key=lambda item: (item[1], item[0]))

        for _, _, nx, ny in candidatos:
            visitados.add((nx, ny))
            ruta.append((nx, ny))
            if backtrack(nx, ny, paso + 1):
                return True
            visitados.remove((nx, ny))
            ruta.pop()
        return False

    return ruta if backtrack(x_ini, y_ini, 1) else None

def resolver_dfs(x_ini, y_ini):
    ruta = [(x_ini, y_ini)]
    visitados = {(x_ini, y_ini)}

    def backtrack(x, y):
        if len(ruta) == 64:
            return True
        
        for nx, ny in obtener_movimientos(x, y):
            if (nx, ny) not in visitados:
                visitados.add((nx, ny))
                ruta.append((nx, ny))
                
                if backtrack(nx, ny):
                    return True
                
                visitados.remove((nx, ny))
                ruta.pop()
        return False

    return ruta if backtrack(x_ini, y_ini) else None
